# Goal Tree Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add parent-child relationships between goals and a collapsible mobile tree view in the manage panel.

**Architecture:** Dexie v3 migration adds `parentGoalId` to Goal. AI prompt generates `parentIndex` per goal. Save logic converts indices to real IDs top-down. A new `GoalTree` component renders the hierarchy as an indented collapsible list. Manage panel gets a List/Tree toggle.

**Tech Stack:** Dexie.js v4 (migration), Zod v4 (schema), Framer Motion (animations), existing locale system

const saveGoalsAndFinish = useCallback(async (data: GenerateGoalsResponse) => {
    try {
    // Save top-down to resolve parentIndex → parentGoalId
    const horizonKeys: GeneratableHorizonKey[] = [
      "lifetime", "10-years", "3-years", "this-year", "this-month",
    ];
    const parentHorizon: Record<string, GeneratableHorizonKey> = {
      "10-years": "lifetime",
      "3-years": "10-years",
      "this-year": "3-years",
      "this-month": "this-year",
    };

    // Map: horizonKey → { goalIndex → savedGoalId }
    const savedIds: Record<string, Record<number, string>> = {};
    let totalGoals = 0;
    let totalTasks = 0;

    for (const hk of horizonKeys) {
      savedIds[hk] = {};
      const goals = data.goals[hk] ?? [];
      for (let i = 0; i < goals.length; i++) {
        const goal = goals[i];
        // Resolve parentGoalId from parentIndex
        let parentGoalId: string | undefined;
        if (goal.parentIndex !== undefined && parentHorizon[hk]) {
          parentGoalId = savedIds[parentHorizon[hk]]?.[goal.parentIndex];
        }

        const goalId = await dataService.addGoal({
          horizonKey: hk,
          title: goal.title,
          description: goal.description,
          priority: goal.priority,
          source: "ai-generated",
          parentGoalId: parentGoalId ? Number(parentGoalId) : undefined,
        });
        savedIds[hk][i] = goalId;
        totalGoals++;

        for (const task of goal.starterTasks) {
          const taskHorizonKey = VALID_HORIZON_KEYS.includes(task.horizonKey as HorizonKey)
            ? (task.horizonKey as HorizonKey)
            : hk;
          await dataService.addTask({
            horizonKey: taskHorizonKey,
            goalId,
            title: task.title,
            estimatedMinutes: task.estimatedMinutes,
            dueDate: task.dueDate,
          });
          totalTasks++;
        }
      }
    }

### Task 5: Tree utility functions

**Files:**
- Create: `src/lib/goal-tree-utils.ts`

- [ ] **Step 1: Create goal-tree-utils.ts**

```typescript
import type { Goal, Task, HorizonKey } from "@/types";

export interface TreeNode {
  goal: Goal;
  children: TreeNode[];
  tasks: Task[];
  depth: number;
}

const HORIZON_ORDER: Record<HorizonKey, number> = {
  "lifetime": 0,
  "10-years": 1,
  "3-years": 2,
  "this-year": 3,
  "this-month": 4,
  "this-week": 5,
};

/**
 * Build a forest (array of root trees) from flat goal + task arrays.
 * Goals without parentGoalId are roots.
 * Tasks attach to their goalId.
 */
export function buildGoalTree(goals: Goal[], tasks: Task[]): TreeNode[] {
  // Index tasks by goalId
  const tasksByGoal = new Map<number, Task[]>();
  for (const t of tasks) {
    if (t.goalId === undefined) continue;
    const gid = typeof t.goalId === "string" ? Number(t.goalId) : t.goalId;
    if (!tasksByGoal.has(gid)) tasksByGoal.set(gid, []);
    tasksByGoal.get(gid)!.push(t);
  }

  // Index goals by id
  const goalMap = new Map<number, Goal>();
  for (const g of goals) {
    if (g.id !== undefined) goalMap.set(g.id, g);
  }

  // Build children map
  const childrenMap = new Map<number, Goal[]>();
  const roots: Goal[] = [];

  for (const g of goals) {
    if (g.parentGoalId !== undefined && goalMap.has(g.parentGoalId)) {
      if (!childrenMap.has(g.parentGoalId)) childrenMap.set(g.parentGoalId, []);
      childrenMap.get(g.parentGoalId)!.push(g);
    } else {
      roots.push(g);
    }
  }

  // Sort roots and children by horizon order
  const sortByHorizon = (a: Goal, b: Goal) =>
    (HORIZON_ORDER[a.horizonKey] ?? 99) - (HORIZON_ORDER[b.horizonKey] ?? 99);

  roots.sort(sortByHorizon);
  for (const children of childrenMap.values()) {
    children.sort(sortByHorizon);
  }

  // Recursively build tree nodes
  function buildNode(goal: Goal, depth: number): TreeNode {
    const children = (childrenMap.get(goal.id!) ?? []).map((c) =>
      buildNode(c, depth + 1)
    );
    const tasks = tasksByGoal.get(goal.id!) ?? [];
    return { goal, children, tasks: tasks.filter((t) => t.status !== "completed"), depth };
  }

  return roots.map((r) => buildNode(r, 0));
}
```

```tsx
"use client";

import { useState } from "react";
import { useLiveQuery } from "dexie-react-hooks";
import { db } from "@/lib/db";
import { buildGoalTree, type TreeNode } from "@/lib/goal-tree-utils";
import { getHorizonColor } from "@/lib/constants";
import { useLocale } from "@/providers/locale-provider";
import { Check, Circle, ChevronDown, ChevronRight, Loader2 } from "lucide-react";
import { dataService } from "@/services";
import { motion, AnimatePresence } from "framer-motion";

function TreeNodeRow({ node }: { node: TreeNode }) {
  const [expanded, setExpanded] = useState(node.depth < 2);
  const { t } = useLocale();
  const hasChildren = node.children.length > 0 || node.tasks.length > 0;
  const indent = node.depth * 16;

  return (
    <div>
      {/* Goal row */}
      <button
        className="flex items-center gap-2 w-full text-left py-2 px-2 rounded-lg hover:bg-gray-50 transition-colors"
        style={{ paddingLeft: indent + 8 }}
        onClick={() => hasChildren && setExpanded(!expanded)}
      >
        {/* Expand/collapse icon */}
        {hasChildren ? (
          expanded ? (
            <ChevronDown className="h-3.5 w-3.5 text-gray-400 shrink-0" />
          ) : (
            <ChevronRight className="h-3.5 w-3.5 text-gray-400 shrink-0" />
          )
        ) : (
          <div className="w-3.5 shrink-0" />
        )}

        {/* Horizon color dot */}
        <div
          className="h-2.5 w-2.5 rounded-full shrink-0"
          style={{ backgroundColor: getHorizonColor(node.goal.horizonKey) }}
        />

        {/* Goal title */}
        <span className="text-sm font-medium text-gray-800 truncate flex-1">
          {node.goal.title}
        </span>

        {/* Horizon label */}
        <span className="text-[10px] text-gray-400 shrink-0">
          {t(`horizon.${node.goal.horizonKey}`)}
        </span>
      </button>

      {/* Children + tasks (collapsible) */}
      <AnimatePresence initial={false}>
        {expanded && (
          <motion.div
            initial={{ height: 0, opacity: 0 }}
            animate={{ height: "auto", opacity: 1 }}
            exit={{ height: 0, opacity: 0 }}
            transition={{ duration: 0.15 }}
            className="overflow-hidden"
          >
            {/* Child goals */}
            {node.children.map((child) => (
              <TreeNodeRow key={child.goal.id} node={child} />
            ))}

            {/* Tasks (only on leaf nodes or nodes with tasks) */}
            {node.tasks.length > 0 && (
              <div style={{ paddingLeft: indent + 40 }}>
                {node.tasks.map((task) => (
                  <button
                    key={task.id}
                    onClick={() =>
                      task.id &&
                      task.status !== "completed" &&
                      dataService.updateTask(String(task.id), {
                        status: "completed",
                        completedAt: new Date(),
                      })
                    }
                    className="flex items-center gap-2 w-full text-left py-1.5 rounded hover:bg-gray-50 transition-colors"
                  >
                    {task.status === "completed" ? (
                      <Check className="h-3.5 w-3.5 text-green-500 shrink-0" />
                    ) : (
                      <Circle className="h-3.5 w-3.5 text-gray-300 shrink-0" />
                    )}
                    <span className="text-xs text-gray-600 truncate">
                      {task.title}
                    </span>
                  </button>
                ))}
              </div>
            )}
          </motion.div>
        )}
      </AnimatePresence>
    </div>
  );
}

export function GoalTree() {
  const { t } = useLocale();
  const goals = useLiveQuery(() => db.goals.toArray(), []);
  const tasks = useLiveQuery(() => db.tasks.toArray(), []);

  if (!goals || !tasks) {
    return (
      <div className="flex justify-center py-8">
        <Loader2 className="h-5 w-5 animate-spin text-gray-400" />
      </div>
    );
  }

  const tree = buildGoalTree(goals, tasks);

  if (tree.length === 0) {
    return (
      <p className="text-sm text-gray-400 text-center py-8">
        {t("tree.noGoals")}
      </p>
    );
  }

  return (
    <div className="space-y-1">
      {tree.map((node) => (
        <TreeNodeRow key={node.goal.id} node={node} />
      ))}
    </div>
  );
}
```
